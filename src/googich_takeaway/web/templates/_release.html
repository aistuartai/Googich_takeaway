{#- One release from the change log: version, date, what it is to this install, and its notes. -#} {%- macro release_date(release) -%} {%- if release.released %}{{ release.released.strftime('%d %B %Y').lstrip('0') }}{% endif -%} {%- endmacro -%} {%- macro release_tags(release) -%} {%- if release.newer %}New{% elif release.running %}Running{% endif -%} {%- if release.candidate %}Release candidate{% endif -%} {%- endmacro -%} {#- The version is a heading at ``level``; Added, Changed and the like sit one level below it. -#} {%- macro release_card(release, level=2) -%}
{{ release.version }} {{ release_date(release) }} {{ release_tags(release) }}
{{ release.notes(level + 1) }}
{%- endmacro -%} {#- A release folded away: version, date and kinds of change, with the notes inside. -#} {%- macro release_folded(release, level=3) -%}
{{ release.version }} {{ release_date(release) }} {{ release_tags(release) }} {% if release.kinds %}{% for kind in release.kinds %}{{ kind }}{% endfor %}{% endif %}
{{ release.notes(level) }}
{%- endmacro -%}
